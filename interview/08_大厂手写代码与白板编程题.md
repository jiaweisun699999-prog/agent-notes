### 题 1：使用 PyTorch / NumPy 手写 Scaled Dot-Product Attention（带 Scaling 和 Causal Mask）

**题目要求**：不使用 `torch.nn.MultiheadAttention`，用 PyTorch 纯张量运算实现自注意力层前向传播，支持 Batch, Head, SeqLen 维度与因果掩码。

**标准解答**：
```python
import math
import torch
import torch.nn as nn
import torch.nn.functional as F

def scaled_dot_product_attention(Q, K, V, mask=None):
    """
    Q: [batch_size, n_heads, seq_len_q, d_k]
    K: [batch_size, n_heads, seq_len_k, d_k]
    V: [batch_size, n_heads, seq_len_k, d_v]
    mask: [batch_size, 1, seq_len_q, seq_len_k] (optional)
    """
    d_k = Q.size(-1)
    
    # 1. 计算点积: [batch_size, n_heads, seq_len_q, seq_len_k]
    scores = torch.matmul(Q, K.transpose(-2, -1)) / math.sqrt(d_k)
    
    # 2. 如果有 Mask (如因果下三角掩码)，将 Mask 标记为 0 的地方设为 -1e9
    if mask is not None:
        scores = scores.masked_fill(mask == 0, -1e9)
        
    # 3. Softmax 归一化
    attn_weights = F.softmax(scores, dim=-1)
    
    # 4. 加权求和输出
    output = torch.matmul(attn_weights, V)
    return output, attn_weights

# 测试代码
if __name__ == "__main__":
    B, H, S, D = 2, 4, 8, 64
    Q = K = V = torch.randn(B, H, S, D)
    # 创建下三角因果掩码
    causal_mask = torch.tril(torch.ones(S, S)).view(1, 1, S, S)
    out, weights = scaled_dot_product_attention(Q, K, V, mask=causal_mask)
    print("Output Shape:", out.shape)  # torch.Size([2, 4, 8, 64])
```

### 题 2：使用 PyTorch 包装手写一个 LoRA Linear 层 (Low-Rank Adaptation)

**题目要求**：继承 `nn.Module`，包裹一个已有的 `nn.Linear` 权重 $W_0$，新增低秩矩阵 $A$ 和 $B$，实现前向传播 $h = W_0 x + \frac{\alpha}{r} B A x$。

**标准解答**：
```python
import math
import torch
import torch.nn as nn

class LoRALinear(nn.Module):
    def __init__(self, base_layer: nn.Linear, r: int = 8, lora_alpha: int = 16):
        super().__init__()
        self.base_layer = base_layer
        # 冻结基础层的权重
        self.base_layer.weight.requires_grad = False
        if self.base_layer.bias is not None:
            self.base_layer.bias.requires_grad = False
            
        in_features = base_layer.in_features
        out_features = base_layer.out_features
        
        self.r = r
        self.lora_alpha = lora_alpha
        self.scaling = lora_alpha / r
        
        # 定义低秩矩阵 A 和 B
        self.lora_A = nn.Parameter(torch.zeros(r, in_features))
        self.lora_B = nn.Parameter(torch.zeros(out_features, r))
        
        # 初始化: A 高斯初始化，B 归零初始化 (确保训练开始时 ΔW = 0)
        nn.init.kaiming_uniform_(self.lora_A, a=math.sqrt(5))
        nn.init.zeros_(self.lora_B)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # 基础层路径
        result = self.base_layer(x)
        # LoRA 路径: x @ A^T @ B^T * scaling
        lora_delta = (x @ self.lora_A.T) @ self.lora_B.T
        return result + lora_delta * self.scaling

# 测试代码
if __name__ == "__main__":
    linear = nn.Linear(512, 1024)
    lora_layer = LoRALinear(linear, r=8, lora_alpha=16)
    x = torch.randn(2, 512)
    output = lora_layer(x)
    print("Output Shape:", output.shape)  # torch.Size([2, 1024])
```

### 题 3：用 Python 手写倒数排名融合算法 (RRF, Reciprocal Rank Fusion)

**题目要求**：给定 Dense 检索和 Sparse 检索返回的两组带文档 ID 的排序列表，实现 RRF 融合并按综合得分从大到小输出 Top-K。

**标准解答**：
```python
from typing import List, Dict

def reciprocal_rank_fusion(
    rankings: List[List[str]], 
    k: int = 60, 
    top_n: int = 5
) -> List[Dict[str, float]]:
    """
    rankings: 包含多个检索器结果的列表，例如 [dense_docs, sparse_docs]
    k: RRF 常数平滑项，默认 60
    top_n: 返回的前 N 个结果
    """
    rrf_scores: Dict[str, float] = {}
    
    for doc_list in rankings:
        for rank, doc_id in enumerate(doc_list, start=1):
            if doc_id not in rrf_scores:
                rrf_scores[doc_id] = 0.0
            # RRF 核心得分累加公式
            rrf_scores[doc_id] += 1.0 / (k + rank)
            
    # 按 RRF 得分降序排序
    sorted_docs = sorted(rrf_scores.items(), key=lambda x: x[1], reverse=True)
    
    return [{"doc_id": doc, "score": score} for doc, score in sorted_docs[:top_n]]

# 测试代码
if __name__ == "__main__":
    dense_results = ["doc_A", "doc_B", "doc_C", "doc_D"]
    sparse_results = ["doc_B", "doc_E", "doc_A", "doc_F"]
    
    fused_results = reciprocal_rank_fusion([dense_results, sparse_results], k=60, top_n=3)
    print("Fused Results:", fused_results)
```

## 三、 Agent & 状态图手写代码题

### 题 4：用 Python 手写一个轻量级 ReAct Agent Loop 引擎框架

**题目要求**：不依赖 LangChain / LlamaIndex，用纯 Python 模拟实现一个包含 Thought, Action, Observation 循环的 Agent 执行引擎。

**标准解答**：
```python
import json

# 1. 模拟工具函数
def get_weather(location: str) -> str:
    return f"{location} 当前天气: 晴朗, 25℃"

TOOLS = {"get_weather": get_weather}

# 2. 模拟大模型 Agent 思考步骤
def mock_llm_react(messages, step):
    if step == 1:
        # 第一步: 输出思考与 Tool 调用动作
        return {
            "thought": "用户询问北京天气，我需要调用 get_weather 工具",
            "action": "get_weather",
            "action_input": {"location": "北京"}
        }
    else:
        # 第二步: 结合观察结果输出最终回答
        return {
            "thought": "我已经获取到了北京的天气信息",
            "final_answer": "北京今天天气晴朗，气温为 25℃，非常舒适。"
        }

# 3. ReAct 引擎主循环
def run_react_agent(user_query: str, max_steps: int = 5):
    messages = [{"role": "user", "content": user_query}]
    
    for step in range(1, max_steps + 1):
        print(f"\n--- [Step {step}] ---")
        response = mock_llm_react(messages, step)
        
        if "final_answer" in response:
            print("🤖 Final Answer:", response["final_answer"])
            return response["final_answer"]
            
        thought = response["thought"]
        tool_name = response["action"]
        tool_input = response["action_input"]
        print(f"🧠 Thought: {thought}")
        print(f"🛠️ Action: {tool_name}({tool_input})")
        
        # 执行工具调用 (Acting)
        if tool_name in TOOLS:
            observation = TOOLS[tool_name](**tool_input)
        else:
            observation = f"Error: Tool {tool_name} 不存在"
            
        print(f"👁️ Observation: {observation}")
        messages.append({"role": "assistant", "content": json.dumps(response)})
        messages.append({"role": "user", "content": f"Observation: {observation}"})

# 测试代码
if __name__ == "__main__":
    run_react_agent("请问北京今天天气怎么样？")
```
